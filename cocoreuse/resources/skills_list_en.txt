# Optimized function for object nnbs
# This function uses the following shapes:
# ['nut', 'nut', 'bridge-v', 'screw']

def nnbs(board, colors, x, y):
    # Define the shapes and their relative positions
    shapes = ['nut', 'nut', 'bridge-v', 'screw']
    # The first instruction places a nut at (1, 8) => (x=0, y=7) relative to (x, y)
    # The second instruction places a nut at (2, 8) => (x=1, y=7) relative to (x, y)  
    # The third instruction places a bridge-v at (1, 8) => (x=0, y=7) relative to (x, y)
    # The fourth instruction places a screw at (2, 8) => (x=1, y=7) relative to (x, y)
    rel_positions = [
        (0, 0),   # nut at (x+0, y+0) = (x, y) = (0, 7) 
        (1, 0),   # nut at (x+1, y+0) = (x+1, y) = (1, 7)
        (0, 0),   # bridge-v at (x+0, y+0) = (x, y) = (0, 7) - covers (0,7) and (1,7)
        (1, 0)    # screw at (x+1, y+0) = (x+1, y) = (1, 7)
    ]
    
    # Place the shapes
    for i, (shape, color, pos) in enumerate(zip(shapes, colors, rel_positions)):
        dx, dy = pos
        put(board, shape, color, x + dx, y + dy)
        
    # Special handling for bridge-v which spans two cells
    # The bridge-v at (0, 7) needs to be placed at (x, y) and (x, y+1)
    # But since we're using relative positions, we need to ensure proper stacking
    # The bridge spans (x, y) and (x, y+1) but we only place it once
    # The nut at (1, 7) should be at (x+1, y) which is (x+1, y)
